//! Spike harness: mesh generators, checks, Monte Carlo, candidate adapters.
use std::collections::HashMap;
use std::f64::consts::PI;

#[derive(Clone, Debug, Default)]
pub struct Mesh {
    pub v: Vec<[f64; 3]>,
    pub t: Vec<[u32; 3]>,
}

#[derive(Clone, Copy, Debug)]
pub enum Op {
    Union,
    Difference,
    Intersection,
}

// ---------- generators (closed, outward CCW, shared vertices) ----------

pub fn make_box(min: [f64; 3], max: [f64; 3]) -> Mesh {
    let mut v = vec![];
    for i in 0..8 {
        v.push([
            if i & 1 == 0 { min[0] } else { max[0] },
            if i & 2 == 0 { min[1] } else { max[1] },
            if i & 4 == 0 { min[2] } else { max[2] },
        ]);
    }
    // vertex index = x + 2y + 4z
    let t = vec![
        [0, 2, 3], [0, 3, 1], // -z
        [4, 5, 7], [4, 7, 6], // +z
        [0, 1, 5], [0, 5, 4], // -y
        [2, 6, 7], [2, 7, 3], // +y
        [0, 4, 6], [0, 6, 2], // -x
        [1, 3, 7], [1, 7, 5], // +x
    ];
    Mesh { v, t }
}

pub const SEG: usize = 64;

/// Cylinder along Z, 64 segments, capped (centre-fan caps).
pub fn make_cylinder(cx: f64, cy: f64, z0: f64, z1: f64, r: f64) -> Mesh {
    let mut v = vec![];
    for i in 0..SEG {
        let a = 2.0 * PI * i as f64 / SEG as f64;
        v.push([cx + r * a.cos(), cy + r * a.sin(), z0]);
    }
    for i in 0..SEG {
        let a = 2.0 * PI * i as f64 / SEG as f64;
        v.push([cx + r * a.cos(), cy + r * a.sin(), z1]);
    }
    let c0 = v.len() as u32;
    v.push([cx, cy, z0]);
    let c1 = v.len() as u32;
    v.push([cx, cy, z1]);
    let mut t = vec![];
    let n = SEG as u32;
    for i in 0..n {
        let j = (i + 1) % n;
        t.push([i, j, n + j]);
        t.push([i, n + j, n + i]);
        t.push([c0, j, i]);
        t.push([c1, n + i, n + j]);
    }
    Mesh { v, t }
}

/// UV sphere, 48 segments x 24 stacks.
pub fn make_sphere(c: [f64; 3], r: f64) -> Mesh {
    let (ns, nst) = (48usize, 24usize);
    let mut v = vec![[c[0], c[1], c[2] + r]]; // north pole = 0
    for k in 1..nst {
        let th = PI * k as f64 / nst as f64;
        for i in 0..ns {
            let ph = 2.0 * PI * i as f64 / ns as f64;
            v.push([c[0] + r * th.sin() * ph.cos(), c[1] + r * th.sin() * ph.sin(), c[2] + r * th.cos()]);
        }
    }
    let south = v.len() as u32;
    v.push([c[0], c[1], c[2] - r]);
    let ring = |k: usize, i: usize| (1 + (k - 1) * ns + i % ns) as u32;
    let mut t = vec![];
    for i in 0..ns {
        t.push([0, ring(1, i), ring(1, i + 1)]);
        for k in 1..nst - 1 {
            t.push([ring(k, i), ring(k + 1, i), ring(k + 1, i + 1)]);
            t.push([ring(k, i), ring(k + 1, i + 1), ring(k, i + 1)]);
        }
        t.push([south, ring(nst - 1, i + 1), ring(nst - 1, i)]);
    }
    Mesh { v, t }
}

/// Swap x and z (winding reversed to keep outward orientation).
pub fn swap_xz(m: &Mesh) -> Mesh {
    Mesh {
        v: m.v.iter().map(|p| [p[2], p[1], p[0]]).collect(),
        t: m.t.iter().map(|t| [t[0], t[2], t[1]]).collect(),
    }
}

pub fn scaled(m: &Mesh, s: f64) -> Mesh {
    Mesh { v: m.v.iter().map(|p| [p[0] * s, p[1] * s, p[2] * s]).collect(), t: m.t.clone() }
}

/// Area of the regular 64-gon of radius r.
pub fn area64(r: f64) -> f64 {
    0.5 * SEG as f64 * r * r * (2.0 * PI / SEG as f64).sin()
}

// ---------- checks ----------

#[derive(Debug, Clone, Default)]
pub struct Report {
    pub closed: bool,
    pub bad_edges: usize,
    pub volume: f64,
    pub degenerate: usize,
    pub tris: usize,
    pub verts: usize,
}

pub fn weld(m: &Mesh, tol: f64) -> Mesh {
    let mut map: HashMap<[i64; 3], u32> = HashMap::new();
    let mut v = vec![];
    let mut remap = vec![0u32; m.v.len()];
    for (i, p) in m.v.iter().enumerate() {
        let k = [(p[0] / tol).round() as i64, (p[1] / tol).round() as i64, (p[2] / tol).round() as i64];
        let id = *map.entry(k).or_insert_with(|| {
            v.push(*p);
            (v.len() - 1) as u32
        });
        remap[i] = id;
    }
    Mesh { v, t: m.t.iter().map(|t| [remap[t[0] as usize], remap[t[1] as usize], remap[t[2] as usize]]).collect() }
}

pub fn analyze(m: &Mesh) -> Report {
    let w = weld(m, 1e-6);
    let mut edges: HashMap<(u32, u32), i32> = HashMap::new(); // directed edge counts
    let mut degenerate = 0;
    let mut vol = 0.0;
    for t in &w.t {
        let (a, b, c) = (w.v[t[0] as usize], w.v[t[1] as usize], w.v[t[2] as usize]);
        let u = [b[0] - a[0], b[1] - a[1], b[2] - a[2]];
        let q = [c[0] - a[0], c[1] - a[1], c[2] - a[2]];
        let n = [u[1] * q[2] - u[2] * q[1], u[2] * q[0] - u[0] * q[2], u[0] * q[1] - u[1] * q[0]];
        let area = 0.5 * (n[0] * n[0] + n[1] * n[1] + n[2] * n[2]).sqrt();
        if area < 1e-12 {
            degenerate += 1;
        }
        vol += (a[0] * (b[1] * c[2] - b[2] * c[1]) - a[1] * (b[0] * c[2] - b[2] * c[0]) + a[2] * (b[0] * c[1] - b[1] * c[0])) / 6.0;
        if t[0] == t[1] || t[1] == t[2] || t[0] == t[2] {
            continue; // collapsed by weld; already counted as degenerate
        }
        for e in [(t[0], t[1]), (t[1], t[2]), (t[2], t[0])] {
            *edges.entry(e).or_insert(0) += 1;
        }
    }
    // closed+manifold: every directed edge used once, and its reverse once
    let mut bad = 0;
    for (&(a, b), &c) in &edges {
        if c != 1 || edges.get(&(b, a)).copied() != Some(1) {
            bad += 1;
        }
    }
    Report { closed: bad == 0 && !w.t.is_empty(), bad_edges: bad, volume: vol, degenerate, tris: w.t.len(), verts: w.v.len() }
}

// ---------- Monte Carlo ----------

pub struct XorShift(pub u64);
impl XorShift {
    pub fn next(&mut self) -> u64 {
        let mut x = self.0;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.0 = x;
        x
    }
    pub fn f(&mut self) -> f64 {
        (self.next() >> 11) as f64 / (1u64 << 53) as f64
    }
}

/// (volume estimate, 1-sigma) of `inside` within box [lo,hi]; 1e6 points, fixed seed.
pub fn monte_carlo(lo: [f64; 3], hi: [f64; 3], inside: impl Fn([f64; 3]) -> bool) -> (f64, f64) {
    let n = 1_000_000u32;
    let mut r = XorShift(0x9E3779B97F4A7C15);
    let mut hit = 0u32;
    for _ in 0..n {
        let p = [lo[0] + (hi[0] - lo[0]) * r.f(), lo[1] + (hi[1] - lo[1]) * r.f(), lo[2] + (hi[2] - lo[2]) * r.f()];
        if inside(p) {
            hit += 1;
        }
    }
    let vb = (hi[0] - lo[0]) * (hi[1] - lo[1]) * (hi[2] - lo[2]);
    let p = hit as f64 / n as f64;
    (p * vb, (p * (1.0 - p) / n as f64).sqrt() * vb)
}

pub fn in_box(p: [f64; 3], lo: [f64; 3], hi: [f64; 3]) -> bool {
    (0..3).all(|i| p[i] >= lo[i] && p[i] <= hi[i])
}

/// Membership in the faceted (64-gon) Z-prism: centre (cx,cy), radius r, z in [z0,z1].
pub fn in_prism64(p: [f64; 3], cx: f64, cy: f64, z0: f64, z1: f64, r: f64) -> bool {
    if p[2] < z0 || p[2] > z1 {
        return false;
    }
    let (dx, dy) = (p[0] - cx, p[1] - cy);
    let apothem = r * (PI / SEG as f64).cos();
    (0..SEG).all(|i| {
        let a = 2.0 * PI * (i as f64 + 0.5) / SEG as f64;
        dx * a.cos() + dy * a.sin() <= apothem
    })
}

// ---------- candidate adapters ----------

pub type R = Result<Mesh, String>;

#[cfg(feature = "boolmesh")]
pub mod cand {
    use super::*;
    pub const NAME: &str = "boolmesh";
    pub fn boolean(op: Op, a: &Mesh, b: &Mesh) -> R {
        use boolmesh::prelude::*;
        let conv = |m: &Mesh| {
            Manifold::new(
                &m.v.iter().flatten().copied().collect::<Vec<f64>>(),
                &m.t.iter().flatten().map(|&i| i as usize).collect::<Vec<_>>(),
            )
        };
        let (ma, mb) = (conv(a)?, conv(b)?);
        let o = match op {
            Op::Union => OpType::Add,
            Op::Difference => OpType::Subtract,
            Op::Intersection => OpType::Intersect,
        };
        let r = compute_boolean(&ma, &mb, o)?;
        let v = r.ps.iter().map(|p| [p.x, p.y, p.z]).collect();
        let idx: Vec<u32> = r.hs.iter().map(|h| h.tail as u32).collect();
        Ok(Mesh { v, t: idx.chunks(3).map(|c| [c[0], c[1], c[2]]).collect() })
    }
}

#[cfg(feature = "manifold-rust")]
pub mod cand {
    use super::*;
    pub const NAME: &str = "manifold-rust";
    pub fn boolean(op: Op, a: &Mesh, b: &Mesh) -> R {
        use manifold_rust::manifold::Manifold;
        use manifold_rust::types::{BooleanEngine, Error, MeshGL64};
        let conv = |m: &Mesh| -> Result<Manifold, String> {
            let g = MeshGL64 {
                num_prop: 3,
                vert_properties: m.v.iter().flatten().copied().collect(),
                tri_verts: m.t.iter().flatten().map(|&i| i as u64).collect(),
                ..Default::default()
            };
            let r = Manifold::from_mesh_gl64(&g);
            if r.status() != Error::NoError {
                return Err(format!("import status {:?}", r.status()));
            }
            Ok(r)
        };
        let (ma, mb) = (conv(a)?, conv(b)?);
        let eng = if std::env::var("MR_ENGINE").as_deref() == Ok("auto") { BooleanEngine::Auto } else { BooleanEngine::Exact };
        let r = match op {
            Op::Union => ma.union_with_engine(&mb, eng),
            Op::Difference => ma.difference_with_engine(&mb, eng),
            Op::Intersection => ma.intersection_with_engine(&mb, eng),
        };
        if r.status() != Error::NoError {
            return Err(format!("result status {:?}", r.status()));
        }
        let g = r.get_mesh_gl64(-1);
        let np = g.num_prop as usize;
        let v = g.vert_properties.chunks(np).map(|c| [c[0], c[1], c[2]]).collect();
        let t = g.tri_verts.chunks(3).map(|c| [c[0] as u32, c[1] as u32, c[2] as u32]).collect();
        Ok(Mesh { v, t })
    }
}

#[cfg(feature = "csgrs")]
pub mod cand {
    use super::*;
    pub const NAME: &str = "csgrs";
    type M = csgrs::mesh::Mesh<()>;
    pub fn boolean(op: Op, a: &Mesh, b: &Mesh) -> R {
        use csgrs::mesh::{polygon::Polygon, vertex::Vertex};
        use csgrs::traits::CSG;
        use nalgebra::Point3;
        let conv = |m: &Mesh| -> M {
            let polys = m
                .t
                .iter()
                .map(|t| {
                    let p = |i: u32| {
                        let q = m.v[i as usize];
                        Point3::new(q[0], q[1], q[2])
                    };
                    let (p0, p1, p2) = (p(t[0]), p(t[1]), p(t[2]));
                    let n = (p1 - p0).cross(&(p2 - p0)).normalize();
                    Polygon::new(vec![Vertex::new(p0, n), Vertex::new(p1, n), Vertex::new(p2, n)], None)
                })
                .collect::<Vec<_>>();
            M::from_polygons(&polys, None)
        };
        let (ma, mb) = (conv(a), conv(b));
        let r = match op {
            Op::Union => ma.union(&mb),
            Op::Difference => ma.difference(&mb),
            Op::Intersection => ma.intersection(&mb),
        };
        let r = r.triangulate();
        let mut out = Mesh::default();
        for p in &r.polygons {
            let base = out.v.len() as u32;
            for v in &p.vertices {
                out.v.push([v.pos.x, v.pos.y, v.pos.z]);
            }
            for k in 1..p.vertices.len() as u32 - 1 {
                out.t.push([base, base + k, base + k + 1]);
            }
        }
        Ok(out)
    }
}

/// Exact union volume of axis-aligned boxes by coordinate compression (cross-check for Monte Carlo).
pub fn exact_box_union_volume(boxes: &[([f64; 3], [f64; 3])]) -> f64 {
    let mut ax: [Vec<f64>; 3] = [vec![], vec![], vec![]];
    for (lo, hi) in boxes {
        for d in 0..3 {
            ax[d].push(lo[d]);
            ax[d].push(hi[d]);
        }
    }
    for a in ax.iter_mut() {
        a.sort_by(|x, y| x.partial_cmp(y).unwrap());
        a.dedup();
    }
    let mut vol = 0.0;
    for i in 0..ax[0].len() - 1 {
        for j in 0..ax[1].len() - 1 {
            for k in 0..ax[2].len() - 1 {
                let c = [(ax[0][i] + ax[0][i + 1]) / 2.0, (ax[1][j] + ax[1][j + 1]) / 2.0, (ax[2][k] + ax[2][k + 1]) / 2.0];
                if boxes.iter().any(|(lo, hi)| in_box(c, *lo, *hi)) {
                    vol += (ax[0][i + 1] - ax[0][i]) * (ax[1][j + 1] - ax[1][j]) * (ax[2][k + 1] - ax[2][k]);
                }
            }
        }
    }
    vol
}

/// Number of unmatched (bad) welded edges that have another welded vertex strictly inside them (T-junctions).
pub fn tjunction_edges(m: &Mesh) -> (usize, usize) {
    let w = weld(m, 1e-6);
    let mut e: HashMap<(u32, u32), i32> = HashMap::new();
    for t in &w.t {
        for k in [(t[0], t[1]), (t[1], t[2]), (t[2], t[0])] {
            if k.0 != k.1 {
                *e.entry(k).or_insert(0) += 1;
            }
        }
    }
    let (mut bad, mut tj) = (0, 0);
    for (&(a, b), &c) in &e {
        if c == 1 && e.get(&(b, a)).copied() == Some(1) {
            continue;
        }
        bad += 1;
        let (pa, pb) = (w.v[a as usize], w.v[b as usize]);
        let d = [pb[0] - pa[0], pb[1] - pa[1], pb[2] - pa[2]];
        let l2 = d[0] * d[0] + d[1] * d[1] + d[2] * d[2];
        let hit = w.v.iter().enumerate().any(|(i, p)| {
            if i as u32 == a || i as u32 == b {
                return false;
            }
            let q = [p[0] - pa[0], p[1] - pa[1], p[2] - pa[2]];
            let s = (q[0] * d[0] + q[1] * d[1] + q[2] * d[2]) / l2;
            if s <= 1e-9 || s >= 1.0 - 1e-9 {
                return false;
            }
            let r = [q[0] - s * d[0], q[1] - s * d[1], q[2] - s * d[2]];
            (r[0] * r[0] + r[1] * r[1] + r[2] * r[2]).sqrt() < 1e-6
        });
        if hit {
            tj += 1;
        }
    }
    (bad, tj)
}
